```javascript
import streamlit as st
import pandas as pd
import plotly.express as px
import plotly.graph_objects as go
import os
import warnings
warnings.filterwarnings('ignore')

st.set_page_config(page_title="사업 현황 대시보드", layout="wide")
st.title("💼 사업 현황 대시보드")

# __file__ 기준 상대경로 (터미널 실행 위치와 무관하게 동작)
BASE_DIR = os.path.dirname(os.path.abspath(__file__))
st.sidebar.write(f"📂 작업 폴더: {BASE_DIR}")

@st.cache_data
def load_data():
    """__file__ 기준 폴더에서 Excel 파일 자동 로드"""

    files = {
        'sales': '사업_월연간목표.xlsx',
        'rate':  '사업_환율연동.xlsx',
        'news':  'business_news.xlsx'
    }

    data = {}

    for key, filename in files.items():
        file_path = os.path.join(BASE_DIR, filename)  # __file__ 기준으로 변경

        if os.path.exists(file_path):
            try:
                data[key] = pd.read_excel(file_path)
                st.sidebar.success(f"✓ {filename}")
            except Exception as e:
                st.sidebar.error(f"❌ {filename}: {e}")
                st.stop()
        else:
            st.sidebar.warning(f"⚠️ {filename} 없음")
            st.stop()

    return data['sales'], data['rate'], data['news']

# 데이터 로드
sales_df, rate_df, news_df = load_data()

# 데이터 타입 변환 (텍스트로 저장된 숫자 대비)
sales_df['실적액'] = pd.to_numeric(sales_df['실적액'], errors='coerce').fillna(0)
sales_df['목표액'] = pd.to_numeric(sales_df['목표액'], errors='coerce').fillna(0)
sales_df['달성률'] = pd.to_numeric(sales_df['달성률'], errors='coerce').fillna(0)
rate_df['환율']   = pd.to_numeric(rate_df['환율'],   errors='coerce').fillna(0)

tab1, tab2, tab3, tab4 = st.tabs(["📊 영업 현황", "💹 환율/뉴스", "📈 분석", "📋 데이터"])

with tab1:
    st.subheader("영업 현황 요약")
    col1, col2, col3, col4 = st.columns(4)
    with col1:
        total = sales_df['실적액'].sum()
        st.metric("총 영업액", f"₩{total/1000000:.0f}M")
    with col2:
        avg = sales_df['달성률'].mean()
        st.metric("평균 달성률", f"{avg:.0f}%")
    with col3:
        st.metric("거래건수", f"{len(sales_df)}건")
    with col4:
        st.metric("담당팀", f"{sales_df['담당팀'].nunique()}팀")

    st.divider()

    col1, col2 = st.columns(2)
    with col1:
        st.subheader("월별 실적 추이")
        monthly = sales_df.groupby('월')['실적액'].sum().reset_index()
        fig1 = px.line(monthly, x='월', y='실적액', title='월별 영업액', markers=True)
        fig1.update_traces(line=dict(color='#FF6B6B', width=3), marker=dict(size=8))
        st.plotly_chart(fig1, use_container_width=True, key="fig1")
    with col2:
        st.subheader("상품별 달성률")
        product = sales_df.groupby('상품')['실적액'].sum().reset_index()
        fig2 = px.pie(product, names='상품', values='실적액', title='상품별 구성비')
        st.plotly_chart(fig2, use_container_width=True, key="fig2")

    st.subheader("팀별 영업액")
    team_sales = sales_df.groupby('담당팀')['실적액'].sum().reset_index()
    fig_team = px.bar(team_sales, x='담당팀', y='실적액', title='담당팀별 영업액',
                      color='실적액', color_continuous_scale='Blues', text='실적액')
    fig_team.update_traces(textposition='outside')
    st.plotly_chart(fig_team, use_container_width=True, key="fig_team")

with tab2:
    st.subheader("환율/뉴스 알림")

    col1, col2, col3 = st.columns(3)
    with col1:
        latest_rate = rate_df['환율'].iloc[0]
        st.metric("현재 환율", f"₩{latest_rate:.0f}")
    with col2:
        latest_change = rate_df['환율변동'].iloc[0] if '환율변동' in rate_df.columns else "N/A"
        st.metric("환율변동", str(latest_change))
    with col3:
        high_news = len(news_df[news_df['중요도'] == '높음'])
        st.metric("높은 중요도", f"{high_news}개")

    st.divider()

    col1, col2 = st.columns([2, 1])
    with col1:
        st.subheader("최근 주요 뉴스 (높은 중요도 우선)")
        high_news_list = news_df[news_df['중요도'] == '높음'].sort_values('날짜', ascending=False).head(10)
        if len(high_news_list) > 0:
            for idx, row in high_news_list.iterrows():
                st.write(f"**{row['키워드']}** / {row['뉴스제목']}")
                st.caption(f"{row['출처']} | {row['날짜']} | {row['영향도']}")
                st.divider()
        else:
            st.info("높은 중요도 뉴스가 없습니다.")
    with col2:
        st.subheader("영향도 분포")
        impact = news_df['영향도'].value_counts().reset_index()
        impact.columns = ['영향도', '개수']
        fig_impact = px.pie(impact, names='영향도', values='개수',
                            color_discrete_map={'긍정': '#2ecc71', '부정': '#e74c3c', '중립': '#95a5a6'},
                            title='영향도 분포')
        st.plotly_chart(fig_impact, use_container_width=True, key="fig_impact")

with tab3:
    st.subheader("분석")

    col1, col2 = st.columns(2)
    with col1:
        st.subheader("팀별 영업액")
        team_sales = sales_df.groupby('담당팀')['실적액'].sum().reset_index()
        fig3 = px.bar(team_sales, x='담당팀', y='실적액', text='실적액',
                      title='담당팀별 영업액',
                      color='실적액', color_continuous_scale='Blues')
        fig3.update_traces(textposition='outside')
        st.plotly_chart(fig3, use_container_width=True, key="fig3")
    with col2:
        st.subheader("환율 추이")
        fig4 = px.line(rate_df, x='날짜', y='환율', title='환율 변동 추이', markers=True)
        fig4.update_traces(line=dict(color='#4ECDC4', width=3), marker=dict(size=8))
        st.plotly_chart(fig4, use_container_width=True, key="fig4")

    st.subheader("중요도별 뉴스 분포")
    importance = news_df['중요도'].value_counts().reset_index()
    importance.columns = ['중요도', '개수']
    fig_imp = px.pie(importance, names='중요도', values='개수', title='중요도 분포')
    st.plotly_chart(fig_imp, use_container_width=True, key="fig_imp")

with tab4:
    st.subheader("📊 월연간 목표 데이터")
    st.dataframe(sales_df, use_container_width=True, hide_index=True)
    st.write(f"**행**: {len(sales_df)}, **컬럼**: {len(sales_df.columns)}")

    st.divider()
    st.subheader("💹 실시간 환율 연동 데이터")
    st.dataframe(rate_df, use_container_width=True, hide_index=True)
    st.write(f"**행**: {len(rate_df)}, **컬럼**: {len(rate_df.columns)}")

    st.divider()
    st.subheader("📰 주요 키워드 뉴스 클리핑 데이터")
    st.dataframe(news_df, use_container_width=True, hide_index=True)
    st.write(f"**행**: {len(news_df)}, **컬럼**: {len(news_df.columns)}")

st.divider()
st.caption("💼 사업 현황 대시보드 | 실시간 모니터링")
```

```plain text
import pandas as pd
import plotly.express as px
import plotly.graph_objects as go
import os
import warnings
warnings.filterwarnings('ignore')

st.set_page_config(page_title="사업 현황 대시보드", layout="wide")
st.title("💼 사업 현황 대시보드")

# 현재 폴더 경로 자동 인식
current_dir = os.getcwd()
st.sidebar.write(f"📂 작업 폴더: {current_dir}")

@st.cache_data

    """현재 폴더에서 Excel 파일 자동 로드"""

    files = {
        'sales': '사업_월연간목표.xlsx',

        'news': 'business_news.xlsx'
    }

    data = {}

    for key, filename in files.items():
        file_path = os.path.join(current_dir, filename)

        if os.path.exists(file_path):
            try:
                data[key] = pd.read_excel(file_path)
                st.sidebar.success(f"✓ {filename}")
            except Exception as e:
                st.sidebar.error(f"❌ {filename}: {e}")
                st.stop()
        else:
            st.sidebar.warning(f"⚠️ {filename} 없음")
            st.stop()

    return data['sales'], data['rate'], data['news']

sales_df['달성률'] = pd.to_numeric(sales_df['달성률'], errors='coerce').fillna(0)

rate_df['환율'] = pd.to_numeric(rate_df['환율'], errors='coerce').fillna(0)

tab1, tab2, tab3, tab4 = st.tabs(["📊 영업 현황", "💹 환율/뉴스", "📈 분석", "📋 데이터"])

    with col1:

        avg = sales_df['달성률'].mean()
        st.metric("평균 달성률", f"{avg:.0f}%")
    with col3:
        count = len(sales_df)
        st.metric("거래건수", f"{count}건")
    with col4:
        unique_team = len(sales_df['담당팀'].unique())
        st.metric("담당팀", f"{unique_team}팀")

    st.divider()

    col1, col2 = st.columns(2)

    with col1:
        st.subheader("월별 실적 추이")

    with col2:

        st.plotly_chart(fig2, use_container_width=True)

    st.subheader("팀별 영업액")
    team_sales = sales_df.groupby('담당팀')['실적액'].sum().reset_index()
    fig_team = px.bar(team_sales, x='담당팀', y='실적액', title='담당팀별 영업액',
                     color='실적액', color_continuous_scale='Blues', text='실적액')
    fig_team.update_traces(textposition='outside')
    st.plotly_chart(fig_team, use_container_width=True)

with tab2:
    st.subheader("환율/뉴스 알림")

    col1, col2, col3 = st.columns(3)
    with col1:
        latest_rate = rate_df['환율'].iloc[0]
        st.metric("현재 환율", f"₩{latest_rate:.0f}")
    with col2:
        latest_change = rate_df['환율변동'].iloc[0] if '환율변동' in rate_df.columns else "N/A"
        st.metric("환율변동", str(latest_change))
    with col3:
        high_news = len(news_df[news_df['중요도'] == '높음'])
        st.metric("높은 중요도", f"{high_news}개")

    st.divider()

    col1, col2 = st.columns([2, 1])

    with col1:

        high_news_list = news_df[news_df['중요도'] == '높음'].sort_values('날짜', ascending=False).head(10)
        if len(high_news_list) > 0:
            for idx, row in high_news_list.iterrows():
                st.write(f"**{row['키워드']}** / {row['뉴스제목']}")

                st.divider()
        else:
            st.info("높은 중요도 뉴스가 없습니다.")

    with col2:
        st.subheader("영향도 분포")
        impact = news_df['영향도'].value_counts().reset_index()
        impact.columns = ['영향도', '개수']
        fig_impact = px.pie(impact, names='영향도', values='개수',
                           color_discrete_map={'긍정': '#2ecc71', '부정': '#e74c3c', '중립': '#95a5a6'},
                           title='영향도 분포')
        st.plotly_chart(fig_impact, use_container_width=True)

with tab3:
    st.subheader("분석")

    col1, col2 = st.columns(2)

    with col1:
        st.subheader("팀별 영업액")
        team_sales = sales_df.groupby('담당팀')['실적액'].sum().reset_index()
        fig3 = px.bar(team_sales, x='담당팀', y='실적액', text='실적액',
                     title='담당팀별 영업액',
                     color='실적액', color_continuous_scale='Blues')
        fig3.update_traces(textposition='outside')
        st.plotly_chart(fig3, use_container_width=True)

    with col2:
        st.subheader("환율 추이")
        fig4 = px.line(rate_df, x='날짜', y='환율', title='환율 변동 추이', markers=True)
        fig4.update_traces(line=dict(color='#4ECDC4', width=3), marker=dict(size=8))
        st.plotly_chart(fig4, use_container_width=True)

    st.subheader("중요도별 뉴스 분포")
    importance = news_df['중요도'].value_counts().reset_index()
    importance.columns = ['중요도', '개수']
    fig_imp = px.pie(importance, names='중요도', values='개수', title='중요도 분포')
    st.plotly_chart(fig_imp, use_container_width=True)

with tab4:
    st.subheader("📊 월연간 목표 데이터")
    st.dataframe(sales_df, use_container_width=True, hide_index=True)
    col1, col2 = st.columns(2)
    with col1:
        st.write(f"**행**: {len(sales_df)}, **컬럼**: {len(sales_df.columns)}")

    st.divider()
    st.subheader("💹 실시간 환율 연동 데이터")
    st.dataframe(rate_df, use_container_width=True, hide_index=True)
    col1, col2 = st.columns(2)
    with col1:
        st.write(f"**행**: {len(rate_df)}, **컬럼**: {len(rate_df.columns)}")

    st.divider()
    st.subheader("📰 주요 키워드 뉴스 클리핑 데이터")
    st.dataframe(news_df, use_container_width=True, hide_index=True)
    col1, col2 = st.columns(2)
    with col1:
        st.write(f"**행**: {len(news_df)}, **컬럼**: {len(news_df.columns)}")

st.divider()
st.caption("💼 사업 현황 대시보드 | 실시간 모니터링")
```
