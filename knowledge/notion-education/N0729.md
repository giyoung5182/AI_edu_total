```plain text
【구조】
날짜, 시간, 라인, 설비, 제품,
목표생산량, 실제생산량, 생산속도,

✓ 시간별 생산량 추이
```

```plain text
【구조】
제품명, 수량, 계획시작일, 계획완료일,
실제시작일, 실제진도율, 상태,
담당라인, 우선순위, 예상완료일

【상태】
예정 (0%) → 진행중 (1~99%) → 완료 (100%)

【의미】
진도율 = 생산량/목표량
지연 = 예상완료일 > 계획완료일

【경영 해석】
진도율 높음 → 납기 안전
지연 예측 → 조치 필요
우선순위 관리 → 리소스 배분
```

```python
import streamlit as st
import pandas as pd
import plotly.express as px
import plotly.graph_objects as go
import os
import warnings
warnings.filterwarnings('ignore')

st.set_page_config(page_title="생산 현황 대시보드", layout="wide")
st.title("🏭 생산 현황 및 일정 대시보드")

# __file__ 기준 상대경로 (터미널 실행 위치와 무관하게 동작)
BASE_DIR = os.path.dirname(os.path.abspath(__file__))
st.sidebar.write(f"📂 작업 폴더: {BASE_DIR}")

@st.cache_data
def load_data():
    """__file__ 기준 폴더에서 Excel 파일 자동 로드"""

    files = {
        'production': '생산_실시간생산.xlsx',
        'schedule':   '생산_생산일정.xlsx'
    }

    data = {}

    for key, filename in files.items():
        file_path = os.path.join(BASE_DIR, filename)  # __file__ 기준으로 변경
        if os.path.exists(file_path):
            try:
                data[key] = pd.read_excel(file_path)
                st.sidebar.success(f"✓ {filename}")
            except Exception as e:
                st.sidebar.error(f"❌ {filename}: {e}")
                st.stop()
        else:
            st.sidebar.warning(f"⚠️ {filename} 없음")
            st.stop()

    return data['production'], data['schedule']

# 데이터 로드
production_df, schedule_df = load_data()

tab1, tab2, tab3, tab4 = st.tabs(["🏭 실시간 생산", "📅 생산 일정", "📊 효율 분석", "📋 데이터"])

with tab1:
    st.subheader("실시간 생산 현황")
    latest_prod = production_df.iloc[-20:].copy()

    col1, col2, col3, col4 = st.columns(4)
    with col1:
        total_output = latest_prod['실제생산량'].sum()
        st.metric("최근 생산량", f"{total_output:.0f}개")
    with col2:
        avg_speed = latest_prod['생산속도'].mean()
        st.metric("평균 생산속도", f"{avg_speed:.1f}개/시")
    with col3:
        avg_defect = latest_prod['불량률'].mean()
        st.metric("평균 불량률", f"{avg_defect:.2f}%")
    with col4:
        avg_utilization = latest_prod['가동률'].mean()
        st.metric("평균 가동률", f"{avg_utilization:.1f}%")

    st.divider()

    col1, col2 = st.columns(2)
    with col1:
        st.subheader("라인별 생산량 현황")
        line_prod = production_df.groupby('라인').agg({
            '실제생산량': 'sum',
            '가동률':     'mean',
            '불량률':     'mean'
        }).reset_index()
        fig1 = px.bar(line_prod, x='라인', y='실제생산량',
                      title='라인별 총 생산량', text='실제생산량',
                      color='불량률', color_continuous_scale='RdYlGn_r')
        fig1.update_traces(textposition='outside')
        st.plotly_chart(fig1, use_container_width=True, key="fig1")
    with col2:
        st.subheader("라인별 불량률")
        fig_defect = px.bar(line_prod, x='라인', y='불량률',
                            title='라인별 평균 불량률',
                            color='불량률', color_continuous_scale='RdYlGn_r')
        st.plotly_chart(fig_defect, use_container_width=True, key="fig_defect")

    st.subheader("시간별 생산 추이")
    fig2 = px.line(production_df, x='시간', y='실제생산량',
                   color='라인', title='시간별 생산량 추이', markers=True)
    fig2.update_traces(marker=dict(size=8))
    st.plotly_chart(fig2, use_container_width=True, key="fig2")

with tab2:
    st.subheader("생산 일정 현황")

    col1, col2, col3, col4 = st.columns(4)
    with col1:
        st.metric("전체 제품", f"{len(schedule_df)}개")
    with col2:
        completed = len(schedule_df[schedule_df['상태'] == '완료'])
        st.metric("완료된 제품", f"{completed}개")
    with col3:
        in_progress = len(schedule_df[schedule_df['상태'] == '진행중'])
        st.metric("진행 중", f"{in_progress}개")
    with col4:
        delayed = len(schedule_df[schedule_df['지연일수'] > 0])
        st.metric("지연 제품", f"{delayed}개")

    st.divider()

    col1, col2 = st.columns(2)
    with col1:
        st.subheader("제품별 진도율")
        fig3 = px.bar(schedule_df.head(15), x='제품명', y='실제진도율',
                      title='제품별 생산 진도율', text='실제진도율',
                      color='실제진도율', color_continuous_scale='RdYlGn')
        fig3.update_traces(textposition='outside')
        st.plotly_chart(fig3, use_container_width=True, key="fig3")
    with col2:
        st.subheader("상태별 제품 분포")
        status_count = schedule_df['상태'].value_counts().reset_index()
        status_count.columns = ['상태', '개수']
        fig4 = px.pie(status_count, names='상태', values='개수',
                      title='생산 상태 분포')
        st.plotly_chart(fig4, use_container_width=True, key="fig4")

    st.subheader("지연 제품 현황")
    delayed_df = schedule_df[schedule_df['지연일수'] > 0].sort_values('지연일수', ascending=False)
    if len(delayed_df) > 0:
        st.dataframe(delayed_df[['제품명', '상태', '실제진도율', '지연일수', '예상완료일']],
                     use_container_width=True, hide_index=True)
    else:
        st.success("✓ 지연 제품 없음")

with tab3:
    st.subheader("생산 효율성 분석")

    col1, col2 = st.columns(2)
    with col1:
        st.subheader("설비별 가동률")
        equipment_stats = production_df.groupby('설비').agg({
            '가동률':     'mean',
            '불량률':     'mean',
            '실제생산량': 'sum'
        }).reset_index()
        fig5 = px.scatter(equipment_stats, x='가동률', y='불량률',
                          size='실제생산량', hover_name='설비',
                          title='설비별 효율성 분석',
                          color='가동률', color_continuous_scale='RdYlGn')
        st.plotly_chart(fig5, use_container_width=True, key="fig5")
    with col2:
        st.subheader("제품별 불량률")
        product_defect = (
            production_df.groupby('제품').agg({
                '불량률':     'mean',
                '실제생산량': 'sum'
            }).reset_index()
            .sort_values('불량률', ascending=False)
            .head(10)
        )
        fig_prod_defect = px.bar(product_defect, x='제품', y='불량률',
                                 title='제품별 불량률 Top 10',
                                 color='불량률', color_continuous_scale='RdYlGn_r')
        st.plotly_chart(fig_prod_defect, use_container_width=True, key="fig_prod_defect")

    st.subheader("목표 대비 실제 생산")
    target_actual = production_df.groupby('라인').agg({
        '목표생산량': 'sum',
        '실제생산량': 'sum'
    }).reset_index()
    fig6 = px.bar(target_actual, x='라인', y=['목표생산량', '실제생산량'],
                  title='라인별 목표 대비 실적', barmode='group')
    st.plotly_chart(fig6, use_container_width=True, key="fig6")

    st.subheader("작업자 수 대비 생산량")
    worker_prod = production_df.groupby('라인').agg({
        '작업자수':   'mean',
        '실제생산량': 'sum'
    }).reset_index()
    worker_prod['생산성'] = worker_prod['실제생산량'] / worker_prod['작업자수']
    fig_worker = px.bar(worker_prod, x='라인', y='생산성',
                        title='라인별 1인당 생산량',
                        color='생산성', color_continuous_scale='Blues')
    st.plotly_chart(fig_worker, use_container_width=True, key="fig_worker")

with tab4:
    st.subheader("🏭 실시간 생산 데이터")
    st.dataframe(production_df, use_container_width=True, hide_index=True)
    st.write(f"**행**: {len(production_df)}, **컬럼**: {len(production_df.columns)}")

    st.divider()
    st.subheader("📅 생산 일정 데이터")
    st.dataframe(schedule_df, use_container_width=True, hide_index=True)
    st.write(f"**행**: {len(schedule_df)}, **컬럼**: {len(schedule_df.columns)}")

st.divider()
st.caption("🏭 생산 현황 및 일정 대시보드 | 실시간 모니터링")
```

```python
# -*- coding: utf-8 -*-
"""
품질 관리 통합 대시보드
- 품질_품질데이터.xlsx  : 라인별 시간대 검사 결과 (불량률, 주요원인)
- 품질_출하일정.xlsx    : 제품별 출하 일정 및 품질 등급
"""

import os
import streamlit as st
import pandas as pd
import plotly.graph_objects as go
import plotly.express as px

# __file__ 기준 상대경로
BASE_DIR      = os.path.dirname(os.path.abspath(__file__))
QUALITY_PATH  = os.path.join(BASE_DIR, "품질_품질데이터.xlsx")
SCHEDULE_PATH = os.path.join(BASE_DIR, "품질_출하일정.xlsx")


# ─────────────────────────────────────────────
# 0. 페이지 설정
# ─────────────────────────────────────────────
st.set_page_config(
    page_title="품질 관리 대시보드",
    page_icon="🔍",
    layout="wide",
)

# 다크테마 config.toml 자동 생성
config_dir  = os.path.join(BASE_DIR, ".streamlit")
config_path = os.path.join(config_dir, "config.toml")
if not os.path.exists(config_path):
    os.makedirs(config_dir, exist_ok=True)
    with open(config_path, "w", encoding="utf-8") as f:
        f.write(
            '[theme]\nbase = "dark"\n'
            'primaryColor = "#3B82F6"\n'
            'backgroundColor = "#0F172A"\n'
            'secondaryBackgroundColor = "#1E293B"\n'
            'textColor = "#F1F5F9"\n'
            'font = "sans serif"\n'
        )

# 공통 CSS
st.markdown("""
<style>
    @import url('[개별 링크 제외]);
    html, body, [class*="css"] { font-family: 'Noto Sans KR', sans-serif; }

    .kpi-card {
        border-radius: 12px; padding: 18px 22px;
        border: 1px solid rgba(255,255,255,0.1);
        border-left: 4px solid;
        background: rgba(255,255,255,0.04);
        margin-bottom: 8px;
    }
    .kpi-label { font-size: 12px; opacity: 0.55; margin-bottom: 5px; }
    .kpi-value { font-size: 26px; font-weight: 700; line-height: 1.2; }
    .kpi-sub   { font-size: 12px; margin-top: 5px; opacity: 0.75; }

    .section-title {
        font-size: 15px; font-weight: 700;
        margin: 24px 0 12px 0; padding-bottom: 7px;
        border-bottom: 1px solid rgba(255,255,255,0.15);
    }
    .badge-ok  { color: #10B981; font-weight: 700; }
    .badge-ng  { color: #EF4444; font-weight: 700; }
    .badge-mid { color: #F59E0B; font-weight: 700; }
</style>
""", unsafe_allow_html=True)

# 1. 데이터 로드

@st.cache_data
def load_data():
    # 품질 검사 데이터
    q = pd.read_excel(QUALITY_PATH)
    q["날짜"] = pd.to_datetime(q["날짜"])
    q["날짜_str"] = q["날짜"].dt.strftime("%Y.%m.%d")

    # 숫자 컬럼 강제 변환 (텍스트로 저장된 경우 대비)
    for col in ["검사수량", "OK수량", "NG수량", "불량률"]:
        if col in q.columns:
            q[col] = pd.to_numeric(q[col], errors="coerce")

    # 출하 일정 데이터
    s = pd.read_excel(SCHEDULE_PATH)
    s["출하계획일"] = pd.to_datetime(s["출하계획일"])
    s["출하예정일"] = pd.to_datetime(s["출하예정일"])

    for col in ["계획수량", "검사완료수량", "검사완료율"]:
        if col in s.columns:
            s[col] = pd.to_numeric(s[col], errors="coerce")

    return q, s

quality_df, schedule_df = load_data()
```

```python
import streamlit as st
import pandas as pd
import plotly.express as px
import plotly.graph_objects as go
import os
import warnings
warnings.filterwarnings('ignore')

st.set_page_config(page_title="생산 현황 대시보드", layout="wide")
st.title("🏭 생산 현황 및 일정 대시보드")

# 현재 폴더 경로 자동 인식
current_dir = os.getcwd()
st.sidebar.write(f"📂 작업 폴더: {current_dir}")

@st.cache_data
def load_data():
    """현재 폴더에서 Excel 파일 자동 로드"""

    files = {
        'production': '생산_실시간생산.xlsx',
        'schedule': '생산_생산일정.xlsx'
    }

    data = {}

    for key, filename in files.items():
        file_path = os.path.join(current_dir, filename)

        # Excel 파일 존재 여부 확인
        if os.path.exists(file_path):
            try:
                data[key] = pd.read_excel(file_path)
                st.sidebar.success(f"✓ {filename}")
            except Exception as e:
                st.sidebar.error(f"❌ {filename}: {e}")
                st.stop()
        else:
            st.sidebar.warning(f"⚠️ {filename} 없음")
            st.stop()

    return data['production'], data['schedule']

# 데이터 로드
production_df, schedule_df = load_data()

tab1, tab2, tab3, tab4 = st.tabs(["🏭 실시간 생산", "📅 생산 일정", "📊 효율 분석", "📋 데이터"])

with tab1:

    latest_prod = production_df.iloc[-20:].copy()

    col1, col2, col3, col4 = st.columns(4)
    with col1:
        total_output = latest_prod['실제생산량'].sum()
        st.metric("최근 생산량", f"{total_output:.0f}개")
    with col2:
        avg_speed = latest_prod['생산속도'].mean()
        st.metric("평균 생산속도", f"{avg_speed:.1f}개/시")
    with col3:
        avg_defect = latest_prod['불량률'].mean()
        st.metric("평균 불량률", f"{avg_defect:.2f}%")
    with col4:
        avg_utilization = latest_prod['가동률'].mean()
        st.metric("평균 가동률", f"{avg_utilization:.1f}%")

    st.divider()

    col1, col2 = st.columns(2)

    with col1:
        st.subheader("라인별 생산량 현황")
        line_prod = production_df.groupby('라인').agg({
            '실제생산량': 'sum',
            '가동률': 'mean',
            '불량률': 'mean'
        }).reset_index()

        fig1 = px.bar(line_prod, x='라인', y='실제생산량',
                      title='라인별 총 생산량', text='실제생산량',
                      color='불량률', color_continuous_scale='RdYlGn_r')
        fig1.update_traces(textposition='outside')
        st.plotly_chart(fig1, use_container_width=True)

    with col2:
        st.subheader("라인별 불량률")
        fig_defect = px.bar(line_prod, x='라인', y='불량률',
                            title='라인별 평균 불량률',
                            color='불량률', color_continuous_scale='RdYlGn_r')
        st.plotly_chart(fig_defect, use_container_width=True)

    st.subheader("시간별 생산 추이")
    fig2 = px.line(production_df, x='시간', y='실제생산량',
                   color='라인', title='시간별 생산량 추이',
                   markers=True)
    fig2.update_traces(marker=dict(size=8))
    st.plotly_chart(fig2, use_container_width=True)

with tab2:
    st.subheader("생산 일정 현황")

    col1, col2, col3, col4 = st.columns(4)
    with col1:
        total_products = len(schedule_df)
        st.metric("전체 제품", f"{total_products}개")

                      title='생산 상태 분포')
        st.plotly_chart(fig4, use_container_width=True)

    st.subheader("지연 제품 현황")
    delayed_df = schedule_df[schedule_df['지연일수'] > 0].sort_values('지연일수', ascending=False)
    if len(delayed_df) > 0:
        st.dataframe(delayed_df[['제품명', '상태', '실제진도율', '지연일수', '예상완료일']],
                     use_container_width=True, hide_index=True)
    else:
        st.success("✓ 지연 제품 없음")

with tab3:
    st.subheader("생산 효율성 분석")

    col1, col2 = st.columns(2)

    with col1:
        st.subheader("설비별 가동률")
        equipment_stats = production_df.groupby('설비').agg({
            '가동률': 'mean',
            '불량률': 'mean',
            '실제생산량': 'sum'
        }).reset_index()

        fig5 = px.scatter(equipment_stats, x='가동률', y='불량률',
                         size='실제생산량', hover_name='설비',
                         title='설비별 효율성 분석',
                         color='가동률', color_continuous_scale='RdYlGn')
        st.plotly_chart(fig5, use_container_width=True)

    with col2:
        st.subheader("제품별 불량률")
        product_defect = production_df.groupby('제품').agg({
            '불량률': 'mean',
            '실제생산량': 'sum'
        }).reset_index().sort_values('불량률', ascending=False).head(10)

        fig_prod_defect = px.bar(product_defect, x='제품', y='불량률',

                                 color='불량률', color_continuous_scale='RdYlGn_r')
        st.plotly_chart(fig_prod_defect, use_container_width=True)

    st.subheader("목표 대비 실제 생산")
    target_actual = production_df.groupby('라인').agg({
        '목표생산량': 'sum',

st.divider()
```

아래 코드를 전체 복사하세요:
```python
import streamlit as st
import pandas as pd
import plotly.express as px
import plotly.graph_objects as go
import os
import warnings
warnings.filterwarnings('ignore')

st.set_page_config(page_title="품질 현황 대시보드", layout="wide")
st.title("✅ 품질 유지 현황 및 특이사항 대시보드")

# 현재 폴더 경로 자동 인식
current_dir = os.getcwd()
st.sidebar.write(f"📂 작업 폴더: {current_dir}")

@st.cache_data
def load_data():
    """현재 폴더에서 Excel 파일 자동 로드"""

    files = {
        'quality': '품질_품질데이터.xlsx',
        'shipment': '품질_출하일정.xlsx'
    }

    data = {}

    for key, filename in files.items():
        file_path = os.path.join(current_dir, filename)

        # Excel 파일 존재 여부 확인
        if os.path.exists(file_path):
            try:
                data[key] = pd.read_excel(file_path)
                st.sidebar.success(f"✓ {filename}")
            except Exception as e:
                st.sidebar.error(f"❌ {filename}: {e}")
                st.stop()
        else:
            st.sidebar.warning(f"⚠️ {filename} 없음")
            st.stop()

    return data['quality'], data['shipment']

# 데이터 로드
quality_df, shipment_df = load_data()

tab1, tab2, tab3, tab4 = st.tabs(["✅ 품질 현황", "📦 출하 일정", "📊 품질 분석", "📋 데이터"])

with tab1:
    st.subheader("실시간 품질 현황")
    latest_quality = quality_df.iloc[-20:].copy()

    col1, col2, col3, col4 = st.columns(4)
    with col1:
        total_inspect = latest_quality['검사수량'].sum()
        st.metric("최근 검사량", f"{total_inspect:.0f}개")
    with col2:
        total_ok = latest_quality['OK수량'].sum()
        st.metric("최근 OK량", f"{total_ok:.0f}개")
    with col3:
        avg_defect = latest_quality['불량률'].mean()
        st.metric("평균 불량률", f"{avg_defect:.2f}%")
    with col4:
        total_ng = latest_quality['NG수량'].sum()
        st.metric("최근 NG량", f"{total_ng:.0f}개")

    st.divider()

    col1, col2 = st.columns(2)

    with col1:
        st.subheader("라인별 품질 현황")
        line_quality = quality_df.groupby('라인').agg({
            '검사수량': 'sum',
            'OK수량': 'sum',
            '불량률': 'mean'
        }).reset_index()

        fig1 = px.bar(line_quality, x='라인', y='불량률',
                      title='라인별 평균 불량률', text='불량률',
                      color='불량률', color_continuous_scale='RdYlGn_r')
        fig1.update_traces(textposition='outside')
        st.plotly_chart(fig1, use_container_width=True)

    with col2:
        st.subheader("라인별 검사수량")
        fig_inspect = px.bar(line_quality, x='라인', y='검사수량',
                            title='라인별 총 검사수량',
                            color='검사수량', color_continuous_scale='Blues')
        st.plotly_chart(fig_inspect, use_container_width=True)

    st.subheader("시간별 불량률 추이")
    fig2 = px.line(quality_df, x='시간', y='불량률',
                   color='라인', title='시간별 불량률 추이',
                   markers=True)
    fig2.update_traces(marker=dict(size=8))
    st.plotly_chart(fig2, use_container_width=True)

with tab2:
    st.subheader("출하 일정 현황")

    col1, col2, col3, col4 = st.columns(4)
    with col1:
        total_products = len(shipment_df)
        st.metric("전체 제품", f"{total_products}개")
    with col2:
        completed = len(shipment_df[shipment_df['상태'] == '출하완료'])
        st.metric("출하완료", f"{completed}개")
    with col3:
        in_progress = len(shipment_df[shipment_df['상태'] == '검사중'])
        st.metric("검사중", f"{in_progress}개")
    with col4:
        ng_products = len(shipment_df[shipment_df['품질등급'] == 'NG'])
        st.metric("품질부적합", f"{ng_products}개")

    st.divider()

    col1, col2 = st.columns(2)

    with col1:
        st.subheader("제품별 검사 완료율")
        fig3 = px.bar(shipment_df.head(15), x='제품명', y='검사완료율',
                      title='제품별 검사 완료율', text='검사완료율',
                      color='검사완료율', color_continuous_scale='RdYlGn')
        fig3.update_traces(textposition='outside')
        st.plotly_chart(fig3, use_container_width=True)

    with col2:
        st.subheader("상태별 제품 분포")
        status_count = shipment_df['상태'].value_counts().reset_index()
        status_count.columns = ['상태', '개수']
        fig4 = px.pie(status_count, names='상태', values='개수',
                      title='출하 상태 분포')
        st.plotly_chart(fig4, use_container_width=True)

    st.subheader("품질등급 분포")
    grade_count = shipment_df['품질등급'].value_counts().reset_index()
    grade_count.columns = ['등급', '개수']
    fig_grade = px.pie(grade_count, names='등급', values='개수',
                       title='품질등급 분포')
    st.plotly_chart(fig_grade, use_container_width=True)

with tab3:
    st.subheader("품질 분석")

    col1, col2 = st.columns(2)

    with col1:
        st.subheader("제품별 불량률")
        product_quality = quality_df.groupby('제품').agg({
            '불량률': 'mean',
            'NG수량': 'sum'
        }).reset_index().sort_values('불량률', ascending=False)

        fig5 = px.bar(product_quality.head(10), x='제품', y='불량률',
                      title='제품별 불량률 (상위 10개)', text='불량률',
                      color='불량률', color_continuous_scale='RdYlGn_r')
        fig5.update_traces(textposition='outside')
        st.plotly_chart(fig5, use_container_width=True)

    with col2:
        st.subheader("불량 원인 분포")
        cause_count = quality_df['주요원인'].value_counts().reset_index().head(10)
        cause_count.columns = ['원인', '발생건수']
        fig6 = px.pie(cause_count, names='원인', values='발생건수',
                      title='불량 원인 TOP10')
        st.plotly_chart(fig6, use_container_width=True)

    st.subheader("OK율 vs 불량률")
    ok_rate = quality_df.groupby('라인').agg({
        'OK수량': 'sum',
        'NG수량': 'sum',
        '검사수량': 'sum'
    }).reset_index()
    ok_rate['OK율'] = (ok_rate['OK수량'] / ok_rate['검사수량'] * 100).round(2)
    ok_rate['불량률'] = (ok_rate['NG수량'] / ok_rate['검사수량'] * 100).round(2)

    fig_rate = px.bar(ok_rate, x='라인', y=['OK율', '불량률'],
                     title='라인별 OK율 vs 불량률',
                     barmode='group')
    st.plotly_chart(fig_rate, use_container_width=True)

with tab4:
    st.subheader("✅ 품질 데이터")
    st.dataframe(quality_df, use_container_width=True, hide_index=True)
    col1, col2 = st.columns(2)
    with col1:
        st.write(f"**행**: {len(quality_df)}, **컬럼**: {len(quality_df.columns)}")

    st.divider()
    st.subheader("📦 출하 일정 데이터")
    st.dataframe(shipment_df, use_container_width=True, hide_index=True)
    col1, col2 = st.columns(2)
    with col1:
        st.write(f"**행**: {len(shipment_df)}, **컬럼**: {len(shipment_df.columns)}")

st.divider()
st.caption("✅ 품질 유지 현황 대시보드 | 실시간 모니터링")
```
