```javascript
import streamlit as st
import pandas as pd
import plotly.express as px
import plotly.graph_objects as go
import os
import warnings
warnings.filterwarnings('ignore')

st.set_page_config(page_title="특허 현황 대시보드", layout="wide")
st.title("🔬 특허 관련 현황 및 키워드 알림 대시보드")

# __file__ 기준 상대경로 (터미널 실행 위치와 무관하게 동작)
BASE_DIR = os.path.dirname(os.path.abspath(__file__))
st.sidebar.write(f"📂 작업 폴더: {BASE_DIR}")

@st.cache_data
def load_data():
    """__file__ 기준 폴더에서 Excel 파일 자동 로드"""

    files = {
        'patent':  '특허_특허현황_데이터.xlsx',
        'keyword': '특허_키워드뉴스_데이터.xlsx'
    }

    data = {}

    for key, filename in files.items():
        file_path = os.path.join(BASE_DIR, filename)  # __file__ 기준으로 변경

        if os.path.exists(file_path):
            try:
                data[key] = pd.read_excel(file_path)
                st.sidebar.success(f"✓ {filename}")
            except Exception as e:
                st.sidebar.error(f"❌ {filename}: {e}")
                st.stop()
        else:
            st.sidebar.warning(f"⚠️ {filename} 없음")
            st.stop()

    return data['patent'], data['keyword']

# 데이터 로드
patent_df, keyword_df = load_data()

tab1, tab2, tab3, tab4 = st.tabs(["🔐 특허 현황", "📰 키워드/뉴스", "📊 분석", "📋 데이터"])

with tab1:
    st.subheader("특허 현황 요약")
    col1, col2, col3, col4, col5 = st.columns(5)
    with col1:
        total = len(patent_df)
        st.metric("총 특허", f"{total}개")
    with col2:
        registered = len(patent_df[patent_df['상태'] == '등록'])
        st.metric("등록", f"{registered}개")
    with col3:
        pending = len(patent_df[patent_df['상태'] == '진행중'])
        st.metric("진행중", f"{pending}개")
    with col4:
        expiring = len(patent_df[patent_df['상태'] == '만료예정'])
        st.metric("만료예정", f"{expiring}개")
    with col5:
        competitors = len(patent_df[patent_df['경쟁사여부'] == 'Yes'])
        st.metric("경쟁사", f"{competitors}개")

    st.divider()

    col1, col2 = st.columns(2)
    with col1:
        st.subheader("기술분야별 특허 분포")
        field = patent_df['기술분야'].value_counts().reset_index()
        field.columns = ['기술분야', '개수']
        fig1 = px.pie(field, names='기술분야', values='개수', title='기술분야별 분포')
        st.plotly_chart(fig1, use_container_width=True, key="fig1")
    with col2:
        st.subheader("특허 상태별 현황")
        status = patent_df['상태'].value_counts().reset_index()
        status.columns = ['상태', '개수']
        fig2 = px.bar(status, x='상태', y='개수', title='상태별 특허 현황', text='개수')
        fig2.update_traces(textposition='outside')
        st.plotly_chart(fig2, use_container_width=True, key="fig2")

    st.subheader("국가별 특허 분포")
    country = patent_df['국가'].value_counts().reset_index()
    country.columns = ['국가', '개수']
    fig_country = px.bar(country, x='국가', y='개수', title='국가별 특허 수', text='개수')
    fig_country.update_traces(textposition='outside')
    st.plotly_chart(fig_country, use_container_width=True, key="fig_country")

with tab2:
    st.subheader("키워드/뉴스 알림")
    col1, col2, col3 = st.columns(3)
    with col1:
        high = len(keyword_df[keyword_df['중요도'] == '높음'])
        st.metric("높음", f"{high}개")
    with col2:
        competitor = len(keyword_df[keyword_df['관련사'] == '경쟁사'])
        st.metric("경쟁사", f"{competitor}개")
    with col3:
        medium = len(keyword_df[keyword_df['중요도'] == '중간'])
        st.metric("중간", f"{medium}개")

    st.divider()

    st.subheader("최근 뉴스 (높은 중요도)")
    high_news = keyword_df[keyword_df['중요도'] == '높음'].sort_values('날짜', ascending=False).head(10)

    if len(high_news) > 0:
        for idx, row in high_news.iterrows():
            col1, col2 = st.columns([4, 1])
            with col1:
                st.write(f"**{row['키워드']}** - {row['뉴스제목']}")
                st.caption(f"{row['출처']} | {row['날짜']}")
            with col2:
                if row['관련사'] == '경쟁사':
                    st.warning("🔴경쟁사")
                else:
                    st.info("🔵협력사")
            st.divider()
    else:
        st.info("높은 중요도 뉴스가 없습니다.")

with tab3:
    col1, col2 = st.columns(2)
    with col1:
        st.subheader("국가별 특허 분포 (상세)")
        country = patent_df['국가'].value_counts().reset_index()
        country.columns = ['국가', '개수']
        fig3 = px.pie(country, names='국가', values='개수', title='국가별 특허 분포')
        st.plotly_chart(fig3, use_container_width=True, key="fig3")
    with col2:
        st.subheader("뉴스 중요도 분포")
        imp = keyword_df['중요도'].value_counts().reset_index()
        imp.columns = ['중요도', '개수']
        fig4 = px.pie(imp, names='중요도', values='개수', title='뉴스 중요도 분포')
        st.plotly_chart(fig4, use_container_width=True, key="fig4")

    st.subheader("키워드별 뉴스 개수")
    keyword_count = keyword_df['키워드'].value_counts().reset_index().head(10)
    keyword_count.columns = ['키워드', '개수']
    fig_keyword = px.bar(keyword_count, x='키워드', y='개수',
                         title='키워드별 뉴스 개수 (상위 10)', text='개수')
    fig_keyword.update_traces(textposition='outside')
    st.plotly_chart(fig_keyword, use_container_width=True, key="fig_keyword")

with tab4:
    st.subheader("🔐 특허 현황 데이터")
    st.dataframe(patent_df, use_container_width=True, hide_index=True)
    st.write(f"**행**: {len(patent_df)}, **컬럼**: {len(patent_df.columns)}")

    st.divider()
    st.subheader("📰 키워드/뉴스 데이터")
    st.dataframe(keyword_df, use_container_width=True, hide_index=True)
    st.write(f"**행**: {len(keyword_df)}, **컬럼**: {len(keyword_df.columns)}")

st.divider()
st.caption("🔬 특허 관련 현황 대시보드 | 실시간 모니터링")
```

```javascript
import streamlit as st
import pandas as pd
import plotly.express as px
import plotly.graph_objects as go
import os
import warnings
warnings.filterwarnings('ignore')

st.set_page_config(page_title="프로젝트 진행현황 대시보드", layout="wide")
st.title("📊 프로젝트 진행현황 대시보드")

# __file__ 기준 상대경로 (터미널 실행 위치와 무관하게 동작)
BASE_DIR = os.path.dirname(os.path.abspath(__file__))
st.sidebar.write(f"📂 작업 폴더: {BASE_DIR}")

@st.cache_data
def load_data():
    """__file__ 기준 폴더에서 Excel 파일 자동 로드"""

    files = {
        'project': '프로젝트진행.xlsx',
        'issue':   'project_issue.xlsx'
    }

    data = {}

    for key, filename in files.items():
        file_path = os.path.join(BASE_DIR, filename)  # __file__ 기준으로 변경

        if os.path.exists(file_path):
            try:
                data[key] = pd.read_excel(file_path)
                st.sidebar.success(f"✓ {filename}")
            except Exception as e:
                st.sidebar.error(f"❌ {filename}: {e}")
                st.stop()
        else:
            st.sidebar.warning(f"⚠️ {filename} 없음")
            st.stop()

    return data['project'], data['issue']

# 데이터 로드
project_df, issue_df = load_data()

tab1, tab2, tab3, tab4 = st.tabs(["📋 프로젝트 현황", "⚠️ 특이사항", "📊 분석", "📋 데이터"])

with tab1:
    st.subheader("프로젝트 현황")
    col1, col2, col3, col4 = st.columns(4)
    with col1:
        st.metric("총 프로젝트", f"{len(project_df)}개")
    with col2:
        avg = project_df['실제진행률'].mean()
        st.metric("평균 진행률", f"{avg:.1f}%")
    with col3:
        high = len(project_df[project_df['위험도'] == '높음'])
        st.metric("고위험도", f"{high}개")
    with col4:
        complete = len(project_df[project_df['실제진행률'] >= 80])
        st.metric("완료 예상", f"{complete}개")

    st.divider()

    col1, col2 = st.columns(2)
    with col1:
        progress_bins = [0, 25, 50, 75, 100]
        project_copy = project_df.copy()
        project_copy['그룹'] = pd.cut(project_copy['실제진행률'], bins=progress_bins)
        progress_dist = project_copy['그룹'].value_counts().sort_index()
        fig1 = px.bar(x=[str(x) for x in progress_dist.index], y=progress_dist.values,
                      title='진행률 분포', labels={'x': '진행률 그룹', 'y': '프로젝트수'},
                      text=progress_dist.values)
        fig1.update_traces(textposition='outside')
        st.plotly_chart(fig1, use_container_width=True, key="fig1")
    with col2:
        team_data = project_df.groupby('담당팀')['프로젝트명'].count().reset_index()
        team_data.columns = ['팀', '수']
        fig2 = px.pie(team_data, names='팀', values='수', title='담당팀 분포')
        st.plotly_chart(fig2, use_container_width=True, key="fig2")

    st.subheader("프로젝트별 진행률")
    progress_sorted = project_df.sort_values('실제진행률', ascending=True).tail(10)
    fig_progress = px.bar(progress_sorted, y='프로젝트명', x='실제진행률',
                          orientation='h', title='프로젝트별 진행률 (상위 10개)',
                          color='실제진행률', color_continuous_scale='RdYlGn')
    st.plotly_chart(fig_progress, use_container_width=True, key="fig_progress")

with tab2:
    st.subheader("특이사항/리스크")
    col1, col2, col3 = st.columns(3)
    with col1:
        high_severity = len(issue_df[issue_df['심각도'] == '높음'])
        st.metric("높음", f"{high_severity}개")
    with col2:
        medium_severity = len(issue_df[issue_df['심각도'] == '중간'])
        st.metric("중간", f"{medium_severity}개")
    with col3:
        low_severity = len(issue_df[issue_df['심각도'] == '낮음'])
        st.metric("낮음", f"{low_severity}개")

    st.divider()

    col1, col2 = st.columns([2, 1])
    with col1:
        st.subheader("높은 심각도 특이사항")
        high_issues = issue_df[issue_df['심각도'] == '높음'].sort_values('날짜', ascending=False).head(10)
        if len(high_issues) > 0:
            for idx, row in high_issues.iterrows():
                st.write(f"**{row['프로젝트명']}** | {row['특이사항']}")
                st.caption(f"{row['담당팀']} | {row['날짜']} | 상태: {row['상태']}")
                st.divider()
        else:
            st.info("높은 심각도 특이사항이 없습니다.")
    with col2:
        severity = issue_df['심각도'].value_counts().reset_index()

        fig_severity = px.pie(severity, names='심각도', values='개수', title='심각도 분포')
        st.plotly_chart(fig_severity, use_container_width=True, key="fig_severity")

with tab3:
    st.subheader("분석")
    col1, col2 = st.columns(2)
    with col1:
        priority_progress = project_df.groupby('우선순위')['실제진행률'].mean().reset_index()

        st.plotly_chart(fig3, use_container_width=True, key="fig3")

    fig_status = px.pie(status_count, names='상태', values='개수', title='진행상태 분포')
    st.plotly_chart(fig_status, use_container_width=True, key="fig_status")

with tab4:
    st.subheader("📋 프로젝트 진행 데이터")
```
