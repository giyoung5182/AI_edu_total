```plain text
Sub 강조_계약조건()

    Dim ws As Worksheet
    Dim lastRow As Long
    Dim i As Long
    Dim 기준날짜 As Date

    Set ws = ThisWorkbook.Sheets("날짜강조")
    lastRow = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row
    기준날짜 = DateSerial(2025, 5, 5)

    For i = 2 To lastRow
        If IsDate(ws.Cells(i, 2).Value) Then
            If ws.Cells(i, 2).Value > 기준날짜 And ws.Cells(i, 5).Value = "진행중" Then
                ws.Rows(i).Interior.Color = RGB(255, 102, 102) ' 연한 빨간색
            End If
        End If
    Next i

    MsgBox "강조 완료!", vbInformation

End Sub
```

```visual basic
Sub 고객사별_계약금액_피벗_및_그래프()

    Dim wsSrc As Worksheet, wsPivot As Worksheet
    Dim pc As PivotCache
    Dim pt As PivotTable
    Dim dataRange As Range
    Dim lastRow As Long
    Dim chartObj As ChartObject

    ' 원본 시트 설정
    Set wsSrc = ThisWorkbook.Sheets("날짜강조")
    lastRow = wsSrc.Cells(wsSrc.Rows.Count, "A").End(xlUp).Row
    Set dataRange = wsSrc.Range("A1").CurrentRegion

    ' 기존 시트 삭제
    On Error Resume Next
    Application.DisplayAlerts = False
    Worksheets("정리요약").Delete
    Application.DisplayAlerts = True
    On Error GoTo 0

    ' 새 시트 생성
    Set wsPivot = Worksheets.Add
    wsPivot.Name = "정리요약"

    ' 피벗 캐시 생성 및 피벗 테이블 생성
    Set pc = ThisWorkbook.PivotCaches.Create(xlDatabase, dataRange)
    Set pt = pc.CreatePivotTable(wsPivot.Range("A3"), "고객사계약피벗")

    ' 피벗 필드 설정
    With pt
        .PivotFields("고객사").Orientation = xlRowField
        .PivotFields("고객사").Position = 1
        .AddDataField .PivotFields("계약금액(만원)"), "총 계약금액", xlSum
        .RowAxisLayout xlTabularRow
    End With

    ' 제목 추가
    With wsPivot.Range("A1")
        .Value = "고객사별 계약금액 요약 (피벗)"
        .Font.Bold = True
        .Font.Size = 16
    End With

    ' 피벗 테이블 서식
    With wsPivot
        .Cells.Font.Size = 12
        .Cells.Font.Name = "맑은 고딕"
        .Columns("A:B").AutoFit
    End With

    ' 차트 생성
    Set chartObj = wsPivot.ChartObjects.Add( _
        Left:=wsPivot.Range("F3").Left, _
        Width:=600, Height:=400, _
        Top:=wsPivot.Range("F3").Top)

    With chartObj.Chart
        .SetSourceData pt.TableRange1
        .ChartType = xlColumnClustered
        .HasTitle = True
        .ChartTitle.Text = "고객사별 계약금액 (만원)"
        .ChartTitle.Font.Size = 14
        .ChartTitle.Font.Bold = True

        With .Axes(xlCategory)
            .HasTitle = True
            .AxisTitle.Text = "고객사"
            .AxisTitle.Font.Size = 12
            .AxisTitle.Font.Bold = True
            .TickLabels.Font.Size = 11
        End With

        With .Axes(xlValue)
            .HasTitle = True
            .AxisTitle.Text = "계약금액(만원)"
            .AxisTitle.Font.Size = 12
            .AxisTitle.Font.Bold = True
            .TickLabels.Font.Size = 11
        End With

        .Legend.Font.Size = 10
    End With

    MsgBox "피벗테이블과 크고 선명한 그래프가 '고객사피벗' 시트에 생성되었습니다!", vbInformation

End Sub
```

```visual basic
Sub 고객사별_계약금액_요약()

    Dim wsSource As Worksheet
    Dim wsDest As Worksheet
    Dim ptCache As PivotCache
    Dim pt As PivotTable
    Dim chartObj As ChartObject
    Dim lastRow As Long

    ' 1. 원본 시트 설정
    Set wsSource = ThisWorkbook.Sheets("날짜강조")
    lastRow = wsSource.Cells(wsSource.Rows.Count, "A").End(xlUp).Row

    ' 2. 기존 "정리요약" 시트 삭제 (있다면)
    On Error Resume Next
    Application.DisplayAlerts = False
    ThisWorkbook.Sheets("정리요약").Delete
    Application.DisplayAlerts = True
    On Error GoTo 0

    ' 3. 새로운 "정리요약" 시트 생성
    Set wsDest = ThisWorkbook.Sheets.Add
    wsDest.Name = "정리요약"

    ' 4. 제목 입력
    wsDest.Range("A1").Value = "고객사별 계약 금액 요약"
    wsDest.Range("A1").Font.Bold = True
    wsDest.Range("A1").Font.Size = 14

    ' 5. 피벗캐시 생성
    Set ptCache = ThisWorkbook.PivotCaches.Create( _
        SourceType:=xlDatabase, _
        SourceData:=wsSource.Name & "!A1:G" & lastRow)

    ' 6. 피벗테이블 생성 (A2 위치)
    Set pt = ptCache.CreatePivotTable( _
        TableDestination:=wsDest.Range("A2"), _
        TableName:="고객사별계약금액요약")

    ' 7. 필드 배치
    With pt
        .PivotFields("고객사").Orientation = xlRowField
        .PivotFields("계약금액(만원)").Orientation = xlDataField
        .PivotFields("계약금액(만원)").Function = xlSum
        .PivotFields("계약금액(만원)").NumberFormat = "#,##0"
        .RowAxisLayout xlTabularRow
    End With

    ' 8. 피벗테이블 오른쪽에 차트 추가 (좌측에 위치시키기 위해 Column 위치 고려)
    Set chartObj = wsDest.ChartObjects.Add(Left:=300, Width:=400, Top:=10, Height:=300)
    With chartObj.Chart
        .SetSourceData Source:=pt.TableRange1
        .ChartType = xlBarClustered
        .HasTitle = True
        .ChartTitle.Text = "고객사별 계약 금액"
    End With

End Sub
```

### 4️⃣ 실습 4: 매크로 버튼 만들기 — 클릭 한 번으로 전부 실행
### 왜 버튼이 필요할까?
지금까지 실습 1\~3에서 VBA 코드를 작성하고, 매번 Visual Basic 편집기를 열어서 실행(▶️) 버튼을 눌렀습니다. 하지만 실무에서는 이렇게 쓰지 않습니다.
실무에서는 **엑셀 시트 위에 버튼을 만들어두고, 클릭 한 번이면 자동으로 실행**되게 합니다. 개발도구를 열 필요도, 코드를 볼 필요도 없습니다. 다른 팀원에게 파일을 넘겨도 "이 버튼 누르면 돼요"라고만 말하면 됩니다.
이번 실습에서는 앞서 만든 기능들을 **버튼 하나로 한꺼번에 실행**하는 매크로 버튼을 만들어 봅니다. 버튼을 클릭하면 날짜 강조 → 피벗테이블+그래프 생성 → 고객사별 파일 분리 저장까지 자동으로 쭉 실행됩니다.
---
### STEP 1: 통합 실행 코드 만들기
앞서 만든 실습 1(날짜 강조), 실습 2(피벗+그래프), 실습 3(고객사별 파일 분리)을 하나의 코드에서 순서대로 실행하는 통합 매크로를 AI에게 요청합니다.
**프롬프트 예시:**
```plain text
앞서 만든 VBA 코드들을 하나로 합쳐서,
아래 작업을 순서대로 실행하는 통합 매크로를 만들어줘.

1. [날짜강조] 시트에서 "계약일"이 2025-5-5 이후이면서
   "진행상태"가 "진행중"인 행을 빨간색으로 강조
2. [날짜강조] 시트의 데이터를 이용해 고객사별 계약금액
   피벗테이블과 막대그래프를 [정리요약] 시트에 생성
3. [날짜강조] 시트의 데이터를 고객사별로 나누어
   각각 별도의 엑셀 파일로 저장
   (저장 경로: [바탕화면의 고객사별_계약 폴더 경로 입력])

모든 작업이 끝나면 "전체 작업이 완료되었습니다!" 라는
메시지 박스를 띄워줘.

매크로 이름은 "전체실행"으로 해줘.
```
> **폴더 경로 넣는 법**: 바탕화면에 "고객사별_계약" 폴더를 만든 뒤, 주소 표시줄을 클릭하여 경로를 복사(Ctrl+C)해서 프롬프트에 붙여넣으면 됩니다.
---
### STEP 2: 엑셀 시트에 버튼 만들기
코드가 준비되었으면, 이제 엑셀 시트 위에 클릭할 수 있는 버튼을 만듭니다.
**버튼 만드는 방법:**
1. 리본 메뉴에서 **개발 도구** 탭 클릭
2. **삽입** → 양식 컨트롤에서 **단추(양식 컨트롤)** 클릭
3. 엑셀 시트 위에서 버튼을 그릴 위치를 드래그
4. "매크로 지정" 창이 나타나면 **"전체실행"** 매크로를 선택하고 확인
5. 버튼 위의 텍스트를 **"자동 실행"** 등 원하는 이름으로 변경

---
### STEP 3: 버튼 클릭으로 실행해 보기
1. 먼저 이전 실습에서 변경된 내용을 초기화합니다 (강조 색상 제거, \[정리요약\] 시트 삭제 등)
2. 만들어둔 **"자동 실행" 버튼**을 클릭합니다
3. 날짜 강조 → 피벗테이블+그래프 생성 → 고객사별 파일 분리 저장이 **자동으로 순서대로 실행**되는 것을 확인합니다
4. "전체 작업이 완료되었습니다!" 메시지가 뜨면 성공
---
### 더 나아가기: 초기화 버튼도 만들어 보기
시간 여유가 있다면, **"초기화" 버튼**도 하나 더 만들어 봅니다. 이 버튼을 누르면 강조 색상이 제거되고, \[정리요약\] 시트가 삭제되어 원래 상태로 돌아갑니다.
**프롬프트 예시:**
```plain text
아래 작업을 수행하는 VBA 코드를 만들어줘. 매크로 이름은 "초기화"로.

1. [날짜강조] 시트의 모든 행 배경색을 흰색으로 초기화
2. [정리요약] 시트가 있으면 삭제
3. [바탕화면의 고객사별_계약 폴더 경로] 안의 파일을 모두 삭제
4. 완료 후 "초기화 완료!" 메시지 박스 표시
```
이 코드를 새 모듈에 붙여넣고, 엑셀 시트에 **"초기화" 버튼**을 하나 더 만들어서 연결합니다.
이제 엑셀 시트 위에 **"자동 실행"** 버튼과 **"초기화"** 버튼 두 개가 나란히 있으면, 누구나 클릭만으로 데이터 강조 → 피벗 보고서 생성 → 고객사별 파일 분리까지 한 번에 처리하고, 되돌릴 수도 있는 자동화 엑셀이 완성됩니다.
---
### 실무 활용 포인트
- 이렇게 버튼이 달린 엑셀 파일을 만들어두면, **매번 반복하는 월간 보고서 작성이나 데이터 정리 작업**을 클릭 한 번으로 처리할 수 있습니다.
- 파일을 팀원에게 넘길 때 "버튼 누르면 됩니다"라고만 전달하면 되므로, VBA를 모르는 사람도 사용할 수 있습니다.
- 버튼이 달린 파일을 저장할 때는 반드시 **"Excel 매크로 사용 통합 문서(.xlsm)"** 형식으로 저장해야 매크로가 유지됩니다.
